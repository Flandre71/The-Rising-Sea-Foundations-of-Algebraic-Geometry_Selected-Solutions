\begin{exercise}
Show that this is the only maximal ideal of $O_{p}$ . (Hint: show that every
element of $O_{p} \setminus m_{p}$ is invertible.)

\begin{proof}
    For $\forall \left( f,U \right) \in \mathscr{O}_{p}\backslash
    \mathfrak{m}_{p}$, we have
    $$
    \left( \frac{1}{f}, U'\right)\in \mathscr{O}_{p}\backslash \mathfrak{m}_{p}
    $$
    in a neighbourhood $U'$ of $p$, hence invertible.
\end{proof}

\end{exercise}

\section{Definition of sheaf and presheaf}

\begin{exercise}[Trivial]
EXERCISE FOR CATEGORY-LOVERS: “A PRESHEAF IS THE SAME AS A CONTRAVARIANT
FUNCTOR”.

Given any topological space X, we have a “category of open sets” (Example
1.1.9), where the objects are the open sets and the morphisms are inclusions.
Verify that the data of a presheaf is precisely the data of a contravariant
functor from the category of open sets of X to the category of sets. (This
interpretation is surprisingly useful.)  
\end{exercise}

\begin{exercise}[Trivial]
IMPORTANT EXERCISE/DEFINITION: THE PUSHFORWARD SHEAF OR DIRECT IMAGE SHEAF.

Suppose $\pi\colon X \to Y$ is a continuous map, and F is a presheaf on X. Then
define a presheaf $\pi_{*}F$ on Y by $\pi_{*}F(V) = \mathcal{F}(\pi^{-1}(V))$ ,
where V is an open subset of Y. Show that $\pi_{*}F$ is a presheaf on Y, and is
a sheaf if F is. This is called the pushforward (or direct image) of F. More
precisely, $\pi_{*}F$ is called the pushforward of F by $\pi$ .
\end{exercise}

\begin{exercise}
(PUSHFORWARD INDUCES MAPS OF STALKS).

Suppose $\pi: X \to Y$ is a continuous map, and $\mathcal{F}$ is a sheaf of sets
(or rings or $\lambda$ -modules) on $X$ . If $\pi(p) = q$ , describe the natural
morphism of stalks $(\pi_*\mathcal{F})_q \to \mathcal{F}_p$ . (You can use the
explicit definition of stalk using representatives, §2.2.4, or the universal
property, §2.2.5. If you prefer one way, you should try the other.)

    Done.
\end{solution}

\section{Morphisms of presheaves and sheaves}

\begin{exercise}
MORPHISMS OF (PRE)SHEAVES INDUCE MORPHISMS OF STALKS.

If $\phi\colon\mathcal{F}\to\mathcal{G}$ is a morphism of presheaves on $X$ ,
and $p\in X$ , describe an induced morphism of stalks
$\phi_{p}\colon\mathcal{F}_{p}\to\mathcal{G}_{p}$ . Translation: taking the
stalk at $p$ induces a stalkification functor $Sets_{X}\to Sets$ . (Your proof
will extend in obvious ways. For example, if $\phi$ is a morphism of
$\mathcal{O}_{X}$ -modules, then $\phi_{p}$ is a map of $\mathcal{O}_{X,p}$
-modules.)

\begin{solution}
    We know elements of stalk $\mathscr{F}_{p}$ consists of $(f,U)$ where $p\in
    U,f\in \mathscr{F}(U)$ and we have $(f,U)\sim(g,V)$ iff
    $$
    \exists W\subseteq U\cap V,f\mid _{W}=g\mid _{W}
    .$$
    Now, consider their images under the presheave morphism
    :
    $$
    \phi:\mathscr{F}\longrightarrow \mathscr{G}
    $$
    $$
    \phi_{U}:\left( f,U \right) \longmapsto \left( \phi\left( f \right) ,U
    \right) 
    $$
    where $\left( \phi\left( f \right) ,U \right) \in \mathscr{G}_{p}$.
    We need to show the induced map
    $$
    \phi_{p}:\mathscr{F}_{p}\longrightarrow \mathscr{G}_{p}
    $$
    is well-defined.

    For
    $\left( f,U \right) \sim\left( g,V \right) $ where $f\mid _{W}=g\mid _{W}$,
    we have
    \begin{align*}
        \phi_{U}\left( f \right)\mid _{W}
        &= \phi_{W}\left( f\mid _{W} \right) \\
        &= \phi_{W}\left( g\mid _{W} \right) \\
        &= \phi_{V}\left( g \right) \mid _{W} \\
    \end{align*}
    Done.
\end{solution}

\end{exercise}

\begin{exercise}
THE COKERNEL DESERVES ITS NAME.

Show that the presheaf cokernel satisfies the universal property of cokernels
(Definition 1.5.4) in the category of presheaves.

\begin{proof}
    By "the universal property of cokernels", we mean there exists unique map
    $$
    \begin{tikzcd}
                                           &                                  & \mathscr{H}                                       \\
    \mathscr{F} \arrow[r] \arrow[rru, "0"] & \mathscr{G} \arrow[r] \arrow[ru] & \coker_{\pre}{\phi} \arrow[u, "\exists!", dotted]
    \end{tikzcd}
    .$$
    Specifically, it means that for any $V\subseteq X$ open, there exists unique
    map
    $$
    \begin{tikzcd}
                                              &                                     & \mathscr{H}(V)                                       \\
    \mathscr{F}(V) \arrow[r] \arrow[rru, "0"] & \mathscr{G}(V) \arrow[r] \arrow[ru] & \coker_{\pre}{\phi(V)} \arrow[u, "\exists!", dotted]
    \end{tikzcd}
    ,$$
    which is already given by the universal property of cokernel
    $\coker_{\pre}{\phi}$ in category of abelian group.
\end{proof}

\end{exercise}

\begin{exercise}[Trivial]
EASY EXERCISE.

Show (or observe) that for a topological space X with open set U, $\mathcal{F}
\mapsto \mathcal{F}(U)$ gives a functor from presheaves of abelian groups on X,
$Ab_{X}^{pre}$ , to abelian groups, Ab. Then show that this functor is exact.

\begin{proof}
    Chase the following diagram, and the result can be obtained immediately.
    $$
    \begin{tikzcd}
    0 \arrow[r] & \mathscr{F} \arrow[r] \arrow[d] & \mathscr{G} \arrow[r] \arrow[d] & \mathscr{H} \arrow[r] \arrow[d] & 0 \\
    0 \arrow[r] & \mathscr{F}{(U)} \arrow[r]      & \mathscr{G}{(U)} \arrow[r]      & \mathscr{H}{(U)} \arrow[r]      & 0
    \end{tikzcd}
    $$
\end{proof}

\end{exercise}
